### 数据库表设计

SRM_MATERIAL_CODING_SERIAL(SRM物料编码规则流水表)

```sql
create table SRM_MATERIAL_CODING_SERIAL
(
    ID     VARCHAR2(32) not null primary key,
    CREATE_BY    VARCHAR2(32),
    CREATE_TIME  DATE,
    UPDATE_BY    VARCHAR2(32),
    UPDATE_TIME  DATE,
    BUSINESS_PATH VARCHAR2(128),
    REAL_PATH VARCHAR2(128),
    INSTANCE_ID VARCHAR2(32),
    VERSION INT,
)
/

comment on table SRM_MATERIAL_CODING_SERIAL is '物料编码流水记录表'
/

comment on column SRM_MATERIAL_CODING_SERIAL.CREATE_BY is '创建人'
/

comment on column SRM_MATERIAL_CODING_SERIAL.CREATE_TIME is '创建日期'
/

comment on column SRM_MATERIAL_CODING_SERIAL.UPDATE_BY is '更新人'
/

comment on column SRM_MATERIAL_CODING_SERIAL.UPDATE_TIME is '更新日期'
/

comment on column SRM_MATERIAL_CODING_SERIAL.BUSINESS_PATH is '业务路径'
/

comment on column SRM_MATERIAL_CODING_SERIAL.REAL_PATH is '真实规则路径'
/

comment on column SRM_MATERIAL_CODING_SERIAL.INSTANCE_ID is '实例记录id'
/

comment on column SRM_MATERIAL_CODING_SERIAL.VERSION is '流水'
```
